62/// Registers every metric this crate declares.
63///
64/// Without this, a metric is only exported after the code path that observes it has run, so a
65/// rarely-taken path leaves its panels blank and makes a routine restart look like the metric
66/// was removed.
67#[cfg(with_metrics)]
68pub fn init_metrics() {
69    linera_base::init_metrics();
70    linera_cache::init_metrics();
71    linera_chain::init_metrics();
72    linera_execution::init_metrics();
73    linera_storage::init_metrics();
74    linera_views::init_metrics();
75    chain_worker::export::metrics::init_metrics();
76    chain_worker::state::metrics::init_metrics();
77    client::metrics::init_metrics();
78    client::requests_scheduler::init_metrics();
79    updater::metrics::init_metrics();
80    worker::metrics::init_metrics();
81}
