49#[expect(clippy::too_many_arguments)]
50pub(crate) async fn forward_cross_chain_queries<F, G>(
51    nickname: String,
52    cross_chain_max_retries: u32,
53    cross_chain_retry_delay: Duration,
54    cross_chain_max_backoff: Duration,
55    cross_chain_sender_delay: Duration,
56    cross_chain_sender_failure_rate: f32,
57    this_shard: ShardId,
58    mut receiver: mpsc::Receiver<(CrossChainRequest, ShardId)>,
59    handle_request: F,
60) where
61    F: Fn(ShardId, CrossChainRequest) -> G + Send + Clone + 'static,
62    G: Future<Output = anyhow::Result<()>> + Send + 'static,
63{
64    let mut steps = tokio::task::JoinSet::new();
65    let mut job_states: HashMap<QueueId, JobState> = HashMap::new();
66
67    let run_task = |task: Task| async move {
68        // Record how long the message waited in queue (in milliseconds)
69        #[cfg(with_metrics)]
70        {
71            let queue_wait_time_ms = task.queued_at.elapsed().as_secs_f64() * 1000.0;
72            metrics::CROSS_CHAIN_QUEUE_WAIT_TIME.observe(queue_wait_time_ms);
73        }
74
75        handle_request(task.shard_id, task.request).await
76    };
77
78    let run_action = |action, queue, state: JobState| async move {
79        linera_base::time::timer::sleep(cross_chain_sender_delay).await;
80
81        let to_shard = state.task.shard_id;
82
83        (
84            queue,
85            match action {
86                Action::Proceed { .. } => {
87                    let target_chain_id = state.task.request.target_chain_id();
88                    if let Err(error) = run_task(state.task).await {
89                        warn!(
90                            nickname = state.nickname,
91                            ?error,
92                            retry = state.retries,
93                            from_shard = this_shard,
94                            to_shard,
95                            chain_id = %target_chain_id,
96                            "Failed to send cross-chain query",
97                        );
98
99                        Action::Retry
100                    } else {
101                        trace!(from_shard = this_shard, to_shard, "Sent cross-chain query",);
102
103                        Action::Proceed {
104                            id: state.id.wrapping_add(1),
105                        }
106                    }
107                }
108
109                Action::Retry => {
110                    let delay = cross_chain_retry_delay
111                        .saturating_mul(state.retries)
112                        .min(cross_chain_max_backoff);
113                    linera_base::time::timer::sleep(delay).await;
114                    Action::Proceed { id: state.id }
115                }
116            },
117        )
118    };
119
120    loop {
121        #[cfg(with_metrics)]
122        #[expect(
123            clippy::cast_possible_wrap,
124            reason = "queue length fits in i64 for any realistic queue size"
125        )]
126        metrics::CROSS_CHAIN_MESSAGE_TASKS.set(job_states.len() as i64);
127
128        tokio::select! {
129            Some(Ok((queue, action))) = steps.join_next() => {
130                let Entry::Occupied(mut state) = job_states.entry(queue) else {
131                    panic!("running job without state");
132                };
133
134                if state.get().is_finished(&action, cross_chain_max_retries) {
135                    state.remove();
136                    continue;
137                }
138
139                if let Action::Retry = action {
140                    state.get_mut().retries += 1
141                }
142
143                steps.spawn(run_action.clone()(action, queue, state.get().clone()));
144            }
145
146            request = receiver.next() => {
147                let Some((request, shard_id)) = request else { break };
148
149                if rand::thread_rng().gen::<f32>() < cross_chain_sender_failure_rate {
150                    warn!("Dropped 1 cross-chain message intentionally.");
151                    continue;
152                }
153
154                let queue = QueueId::new(&request);
155
156                let task = Task {
157                    shard_id,
158                    request,
159                    #[cfg(with_metrics)]
160                    queued_at: Instant::now(),
161                };
162
163                match job_states.entry(queue) {
164                    Entry::Vacant(entry) => {
165                        steps.spawn(run_action.clone()(
166                            Action::Proceed { id: 0 },
167                            queue,
168                            entry.insert(JobState {
169                                id: 0,
170                                retries: 0,
171                                nickname: nickname.clone(),
172                                task,
173                            }).clone(),
174                        ));
175                    }
176
177                    Entry::Occupied(mut entry) => {
178                        entry.insert(JobState {
179                            id: entry.get().id + 1,
180                            retries: 0,
181                            nickname: nickname.clone(),
182                            task,
183                        });
184                    }
185                }
186            }
187
188            else => (),
189        }
190    }
191}

193/// An discriminant for message queues: messages with the same queue ID will be delivered
194/// in order.
195#[derive(Copy, Clone, PartialEq, Eq, Hash)]
196struct QueueId {
197    sender: ChainId,
198    recipient: ChainId,
199    is_update: bool,
200}
201
202impl QueueId {
203    /// Returns a discriminant for the message's queue.
204    fn new(request: &CrossChainRequest) -> Self {
205        let (sender, recipient, is_update) = match request {
206            CrossChainRequest::UpdateRecipient {
207                sender, recipient, ..
208            } => (*sender, *recipient, true),
209            CrossChainRequest::ConfirmUpdatedRecipient {
210                sender, recipient, ..
211            }
212            | CrossChainRequest::RevertConfirm {
213                sender, recipient, ..
214            } => (*sender, *recipient, false),
215        };
216        QueueId {
217            sender,
218            recipient,
219            is_update,
220        }
221    }
222}

224enum Action {
225    /// The request has been sent successfully and the next request can be sent.
226    Proceed { id: usize },
227    /// The request failed and should be retried.
228    Retry,
229}
230
231#[derive(Clone)]
232struct Task {
233    /// The ID of the shard the request is sent to.
234    pub shard_id: ShardId,
235    /// The cross-chain request to be sent.
236    pub request: linera_core::data_types::CrossChainRequest,
237    /// When this task was queued.
238    #[cfg(with_metrics)]
239    pub queued_at: Instant,
240}
241
242#[derive(Clone)]
243struct JobState {
244    /// Queued requests are assigned incremental IDs.
245    pub id: usize,
246    /// How often the current request has been retried.
247    pub retries: u32,
248    /// The nickname of this worker, i.e. the one that is sending the request.
249    pub nickname: String,
250    /// The current request to be sent.
251    pub task: Task,
252}
253
254impl JobState {
255    /// Returns whether the job is finished and should be removed.
256    fn is_finished(&self, action: &Action, max_retries: u32) -> bool {
257        match action {
258            // If the action is to proceed and no new messages with a higher ID are waiting.
259            Action::Proceed { id } => self.id < *id,
260            // If the action is to retry and the maximum number of retries has been reached.
261            Action::Retry => self.retries >= max_retries,
262        }
263    }
264}

325    /// A `handle_request` implementation that records the requests it is given and decides
326    /// their outcome.
327    #[derive(Clone)]
328    struct Handler {
329        calls: Arc<Mutex<Vec<(ShardId, CrossChainRequest)>>>,
330        /// How many further calls fail before they start succeeding.
331        failures_left: Arc<AtomicUsize>,
332        /// If set, each call takes one permit before returning, so that a test can hold
333        /// requests in flight.
334        gate: Option<Arc<Semaphore>>,
335        /// Emits one message per call, so that [`Handler::wait_for_calls`] can await
336        /// instead of spinning.
337        call_signal: UnboundedSender<()>,
338        call_signals: Arc<AsyncMutex<UnboundedReceiver<()>>>,
339    }
340
341    impl Handler {
342        fn new() -> Self {
343            let (call_signal, call_signals) = unbounded_channel();
344            Handler {
345                calls: Arc::default(),
346                failures_left: Arc::new(AtomicUsize::new(0)),
347                gate: None,
348                call_signal,
349                call_signals: Arc::new(AsyncMutex::new(call_signals)),
350            }
351        }
352
353        /// Makes the next `count` calls fail.
354        fn failing(self, count: usize) -> Self {
355            self.failures_left.store(count, Ordering::SeqCst);
356            self
357        }
358
359        /// Makes every call block until the test releases it with [`Handler::release`].
360        fn gated(mut self) -> Self {
361            self.gate = Some(Arc::new(Semaphore::new(0)));
362            self
363        }
364
365        /// Lets `count` blocked calls return.
366        fn release(&self, count: usize) {
367            self.gate
368                .as_ref()
369                .expect("handler is gated")
370                .add_permits(count);
371        }
372
373        /// The closure to hand to [`forward_cross_chain_queries`].
374        fn as_fn(
375            &self,
376        ) -> impl Fn(ShardId, CrossChainRequest) -> BoxFuture<'static, anyhow::Result<()>>
377               + Send
378               + Clone
379               + 'static {
380            let this = self.clone();
381            move |shard_id, request| {
382                let this = this.clone();
383                Box::pin(async move { this.call(shard_id, request).await })
384            }
385        }
386
387        async fn call(self, shard_id: ShardId, request: CrossChainRequest) -> anyhow::Result<()> {
388            self.calls.lock().unwrap().push((shard_id, request));
389            // A closed receiver just means the test has finished waiting for calls.
390            self.call_signal.send(()).ok();
391            if let Some(gate) = &self.gate {
392                gate.acquire()
393                    .await
394                    .expect("the gate is never closed")
395                    .forget();
396            }
397            // `checked_sub` yields `None` once the budget is exhausted, which makes
398            // `fetch_update` fail without touching the counter.
399            let fails = self
400                .failures_left
401                .fetch_update(Ordering::SeqCst, Ordering::SeqCst, |left| {
402                    left.checked_sub(1)
403                })
404                .is_ok();
405            if fails {
406                anyhow::bail!("simulated transport failure");
407            }
408            Ok(())
409        }
410
411        /// Waits until `handle_request` has been called `count` times in total, panicking if
412        /// that many calls never arrive.
413        ///
414        /// Awaiting rather than polling in a loop is what makes the timing assertions work:
415        /// the paused clock only advances while every task is idle, so a spinning test would
416        /// stop time and hang. The [`SETTLE`] bound is simulated time too, so a forwarder
417        /// that stops making calls fails the test instead of hanging it.
418        async fn wait_for_calls(&self, count: usize) {
419            let wait = async {
420                let mut signals = self.call_signals.lock().await;
421                while self.call_count() < count {
422                    signals.recv().await.expect("the handler outlives the test");
423                }
424            };
425            linera_base::time::timer::timeout(SETTLE, wait)
426                .await
427                .unwrap_or_else(|_| {
428                    panic!(
429                        "expected {count} calls, but the forwarder stopped after {}",
430                        self.call_count()
431                    )
432                });
433        }
434
435        fn calls(&self) -> Vec<(ShardId, CrossChainRequest)> {
436            self.calls.lock().unwrap().clone()
437        }
438
439        fn call_count(&self) -> usize {
440            self.calls.lock().unwrap().len()
441        }
442
443        /// The `tag` of each delivered [`confirm`] request, in delivery order.
444        fn tags(&self) -> Vec<u64> {
445            self.calls()
446                .into_iter()
447                .map(|(_, request)| match request {
448                    CrossChainRequest::ConfirmUpdatedRecipient { latest_height, .. } => {
449                        latest_height.0
450                    }
451                    other => panic!("not a confirmation: {other:?}"),
452                })
453                .collect()
454        }
455    }

655    /// The wait before the n-th retry is `retry_delay * n`, capped at `max_backoff`.
656    #[tokio::test(start_paused = true)]
657    async fn test_retry_delay_grows_linearly_up_to_the_maximum() {
658        let handler = Handler::new().failing(usize::MAX);
659        let (mut sender, _task) = spawn_forwarder(
660            &handler,
661            4,
662            Duration::from_secs(10),
663            Duration::from_secs(25),
664            NO_DELAY,
665            NEVER_DROP,
666        );
667
668        let start = tokio::time::Instant::now();
669        send(&mut sender, confirm(1, 2, 1), 0).await;
670
671        let mut attempt_times = Vec::new();
672        for attempt in 1..=5 {
673            handler.wait_for_calls(attempt).await;
674            attempt_times.push(start.elapsed());
675        }
676        settle().await;
677
678        assert_eq!(handler.call_count(), 5, "one attempt plus four retries");
679        // Waits of 10s, 20s, 25s (capped) and 25s between consecutive attempts.
680        assert_eq!(
681            attempt_times,
682            vec![
683                Duration::ZERO,
684                Duration::from_secs(10),
685                Duration::from_secs(30),
686                Duration::from_secs(55),
