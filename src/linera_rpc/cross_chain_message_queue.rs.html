48#[expect(clippy::too_many_arguments)]
49pub(crate) async fn forward_cross_chain_queries<F, G>(
50    nickname: String,
51    cross_chain_max_retries: u32,
52    cross_chain_retry_delay: Duration,
53    cross_chain_max_backoff: Duration,
54    cross_chain_sender_delay: Duration,
55    cross_chain_sender_failure_rate: f32,
56    this_shard: ShardId,
57    mut receiver: mpsc::Receiver<(CrossChainRequest, ShardId)>,
58    handle_request: F,
59) where
60    F: Fn(ShardId, CrossChainRequest) -> G + Send + Clone + 'static,
61    G: Future<Output = anyhow::Result<()>> + Send + 'static,
62{
63    let mut steps = tokio::task::JoinSet::new();
64    let mut job_states: HashMap<QueueId, JobState> = HashMap::new();
65
66    let run_task = |task: Task| async move {
67        // Record how long the message waited in queue (in milliseconds)
68        #[cfg(with_metrics)]
69        {
70            let queue_wait_time_ms = task.queued_at.elapsed().as_secs_f64() * 1000.0;
71            metrics::CROSS_CHAIN_QUEUE_WAIT_TIME.observe(queue_wait_time_ms);
72        }
73
74        handle_request(task.shard_id, task.request).await
75    };
76
77    let run_action = |action, queue, state: JobState| async move {
78        linera_base::time::timer::sleep(cross_chain_sender_delay).await;
79
80        let to_shard = state.task.shard_id;
81
82        (
83            queue,
84            match action {
85                Action::Proceed { .. } => {
86                    let target_chain_id = state.task.request.target_chain_id();
87                    if let Err(error) = run_task(state.task).await {
88                        warn!(
89                            nickname = state.nickname,
90                            ?error,
91                            retry = state.retries,
92                            from_shard = this_shard,
93                            to_shard,
94                            chain_id = %target_chain_id,
95                            "Failed to send cross-chain query",
96                        );
97
98                        Action::Retry
99                    } else {
100                        trace!(from_shard = this_shard, to_shard, "Sent cross-chain query",);
101
102                        Action::Proceed {
103                            id: state.id.wrapping_add(1),
104                        }
105                    }
106                }
107
108                Action::Retry => {
109                    let delay = cross_chain_retry_delay
110                        .saturating_mul(state.retries)
111                        .min(cross_chain_max_backoff);
112                    linera_base::time::timer::sleep(delay).await;
113                    Action::Proceed { id: state.id }
114                }
115            },
116        )
117    };
118
119    // Every step must report back the queue it belongs to. A step that ends in a panic
120    // reports nothing: `join_next` yields a `JoinError` instead of a `(QueueId, Action)`
121    // pair, so the queue keeps its entry in `job_states` with no running step, and every
122    // later request for it is only recorded there and never sent — that queue stops
123    // delivering for the lifetime of the process. Catching the panic turns it into an
124    // ordinary failed attempt, leaving the queue under the usual retry and give-up logic.
125    let run_action = move |action, queue: QueueId, state: JobState| {
126        // Kept out of `state`, which the step consumes, so that the panic can be reported
127        // with the same fields as an ordinary send failure.
128        let nickname = state.nickname.clone();
129        let to_shard = state.task.shard_id;
130        let retries = state.retries;
131        let step = run_action.clone()(action, queue, state);
132        async move {
133            AssertUnwindSafe(step)
134                .catch_unwind()
135                .await
136                .unwrap_or_else(|_| {
137                    error!(
138                        nickname,
139                        retry = retries,
140                        from_shard = this_shard,
141                        to_shard,
142                        sender = %queue.sender,
143                        recipient = %queue.recipient,
144                        "Panic while sending a cross-chain query; treating it as a failed attempt",
145                    );
146                    (queue, Action::Retry)
147                })
148        }
149    };
150
151    loop {
152        #[cfg(with_metrics)]
153        #[expect(
154            clippy::cast_possible_wrap,
155            reason = "queue length fits in i64 for any realistic queue size"
156        )]
157        metrics::CROSS_CHAIN_MESSAGE_TASKS.set(job_states.len() as i64);
158
159        tokio::select! {
160            Some(Ok((queue, action))) = steps.join_next() => {
161                let Entry::Occupied(mut state) = job_states.entry(queue) else {
162                    panic!("running job without state");
163                };
164
165                if state.get().is_finished(&action, cross_chain_max_retries) {
166                    state.remove();
167                    continue;
168                }
169
170                if let Action::Retry = action {
171                    state.get_mut().retries += 1
172                }
173
174                steps.spawn(run_action.clone()(action, queue, state.get().clone()));
175            }
176
177            request = receiver.next() => {
178                let Some((request, shard_id)) = request else { break };
179
180                if rand::thread_rng().gen::<f32>() < cross_chain_sender_failure_rate {
181                    warn!("Dropped 1 cross-chain message intentionally.");
182                    continue;
183                }
184
185                let queue = QueueId::new(&request);
186
187                let task = Task {
188                    shard_id,
189                    request,
190                    #[cfg(with_metrics)]
191                    queued_at: Instant::now(),
192                };
193
194                match job_states.entry(queue) {
195                    Entry::Vacant(entry) => {
196                        steps.spawn(run_action.clone()(
197                            Action::Proceed { id: 0 },
198                            queue,
199                            entry.insert(JobState {
200                                id: 0,
201                                retries: 0,
202                                nickname: nickname.clone(),
203                                task,
204                            }).clone(),
205                        ));
206                    }
207
208                    Entry::Occupied(mut entry) => {
209                        entry.insert(JobState {
210                            id: entry.get().id + 1,
211                            retries: 0,
212                            nickname: nickname.clone(),
213                            task,
214                        });
215                    }
216                }
217            }
218
219            else => (),
220        }
221    }
222}

224/// An discriminant for message queues: messages with the same queue ID will be delivered
225/// in order.
226#[derive(Copy, Clone, PartialEq, Eq, Hash)]
227struct QueueId {
228    sender: ChainId,
229    recipient: ChainId,
230    is_update: bool,
231}
232
233impl QueueId {
234    /// Returns a discriminant for the message's queue.
235    fn new(request: &CrossChainRequest) -> Self {
236        let (sender, recipient, is_update) = match request {
237            CrossChainRequest::UpdateRecipient {
238                sender, recipient, ..
239            } => (*sender, *recipient, true),
240            CrossChainRequest::ConfirmUpdatedRecipient {
241                sender, recipient, ..
242            }
243            | CrossChainRequest::RevertConfirm {
244                sender, recipient, ..
245            } => (*sender, *recipient, false),
246        };
247        QueueId {
248            sender,
249            recipient,
250            is_update,
251        }
252    }
253}

255enum Action {
256    /// The request has been sent successfully and the next request can be sent.
257    Proceed { id: usize },
258    /// The request failed and should be retried.
259    Retry,
260}

262#[derive(Clone)]
263struct Task {
264    /// The ID of the shard the request is sent to.
265    pub shard_id: ShardId,
266    /// The cross-chain request to be sent.
267    pub request: linera_core::data_types::CrossChainRequest,
268    /// When this task was queued.
269    #[cfg(with_metrics)]
270    pub queued_at: Instant,
271}

273#[derive(Clone)]
274struct JobState {
275    /// Queued requests are assigned incremental IDs.
276    pub id: usize,
277    /// How often the current request has been retried.
278    pub retries: u32,
279    /// The nickname of this worker, i.e. the one that is sending the request.
280    pub nickname: String,
281    /// The current request to be sent.
282    pub task: Task,
283}
284
285impl JobState {
286    /// Returns whether the job is finished and should be removed.
287    fn is_finished(&self, action: &Action, max_retries: u32) -> bool {
288        match action {
289            // If the action is to proceed and no new messages with a higher ID are waiting.
290            Action::Proceed { id } => self.id < *id,
291            // If the action is to retry and the maximum number of retries has been reached.
292            Action::Retry => self.retries >= max_retries,
293        }
294    }
295}

356    /// A `handle_request` implementation that records the requests it is given and decides
357    /// their outcome.
358    #[derive(Clone)]
359    struct Handler {
360        calls: Arc<Mutex<Vec<(ShardId, CrossChainRequest)>>>,
361        /// How many further calls fail before they start succeeding.
362        failures_left: Arc<AtomicUsize>,
363        /// How many further calls panic before they stop panicking.
364        panics_left: Arc<AtomicUsize>,
365        /// If set, each call takes one permit before returning, so that a test can hold
366        /// requests in flight.
367        gate: Option<Arc<Semaphore>>,
368        /// Emits one message per call, so that [`Handler::wait_for_calls`] can await
369        /// instead of spinning.
370        call_signal: UnboundedSender<()>,
371        call_signals: Arc<AsyncMutex<UnboundedReceiver<()>>>,
372    }
373
374    impl Handler {
375        fn new() -> Self {
376            let (call_signal, call_signals) = unbounded_channel();
377            Handler {
378                calls: Arc::default(),
379                failures_left: Arc::new(AtomicUsize::new(0)),
380                panics_left: Arc::new(AtomicUsize::new(0)),
381                gate: None,
382                call_signal,
383                call_signals: Arc::new(AsyncMutex::new(call_signals)),
384            }
385        }
386
387        /// Makes the next `count` calls fail.
388        fn failing(self, count: usize) -> Self {
389            self.failures_left.store(count, Ordering::SeqCst);
390            self
391        }
392
393        /// Makes the next `count` calls panic.
394        fn panicking(self, count: usize) -> Self {
395            self.panics_left.store(count, Ordering::SeqCst);
396            self
397        }
398
399        /// Makes every call block until the test releases it with [`Handler::release`].
400        fn gated(mut self) -> Self {
401            self.gate = Some(Arc::new(Semaphore::new(0)));
402            self
403        }
404
405        /// Lets `count` blocked calls return.
406        fn release(&self, count: usize) {
407            self.gate
408                .as_ref()
409                .expect("handler is gated")
410                .add_permits(count);
411        }
412
413        /// The closure to hand to [`forward_cross_chain_queries`].
414        fn as_fn(
415            &self,
416        ) -> impl Fn(ShardId, CrossChainRequest) -> BoxFuture<'static, anyhow::Result<()>>
417               + Send
418               + Clone
419               + 'static {
420            let this = self.clone();
421            move |shard_id, request| {
422                let this = this.clone();
423                Box::pin(async move { this.call(shard_id, request).await })
424            }
425        }
426
427        async fn call(self, shard_id: ShardId, request: CrossChainRequest) -> anyhow::Result<()> {
428            self.calls.lock().unwrap().push((shard_id, request));
429            // A closed receiver just means the test has finished waiting for calls.
430            self.call_signal.send(()).ok();
431            if let Some(gate) = &self.gate {
432                gate.acquire()
433                    .await
434                    .expect("the gate is never closed")
435                    .forget();
436            }
437            // `checked_sub` yields `None` once a budget is exhausted, which makes
438            // `fetch_update` fail without touching the counter.
439            let take = |budget: &AtomicUsize| {
440                budget
441                    .fetch_update(Ordering::SeqCst, Ordering::SeqCst, |left| {
442                        left.checked_sub(1)
443                    })
444                    .is_ok()
445            };
446            if take(&self.panics_left) {
447                panic!("simulated transport panic");
448            }
449            if take(&self.failures_left) {
450                anyhow::bail!("simulated transport failure");
451            }
452            Ok(())
453        }
454
455        /// Waits until `handle_request` has been called `count` times in total, panicking if
456        /// that many calls never arrive.
457        ///
458        /// Awaiting rather than polling in a loop is what makes the timing assertions work:
459        /// the paused clock only advances while every task is idle, so a spinning test would
460        /// stop time and hang. The [`SETTLE`] bound is simulated time too, so a forwarder
461        /// that stops making calls fails the test instead of hanging it.
462        async fn wait_for_calls(&self, count: usize) {
463            let wait = async {
464                let mut signals = self.call_signals.lock().await;
465                while self.call_count() < count {
466                    signals.recv().await.expect("the handler outlives the test");
467                }
468            };
469            linera_base::time::timer::timeout(SETTLE, wait)
470                .await
471                .unwrap_or_else(|_| {
472                    panic!(
473                        "expected {count} calls, but the forwarder stopped after {}",
474                        self.call_count()
475                    )
476                });
477        }
478
479        fn calls(&self) -> Vec<(ShardId, CrossChainRequest)> {
480            self.calls.lock().unwrap().clone()
481        }
482
483        fn call_count(&self) -> usize {
484            self.calls.lock().unwrap().len()
485        }
486
487        /// The `tag` of each delivered [`confirm`] request, in delivery order.
488        fn tags(&self) -> Vec<u64> {
489            self.calls()
490                .into_iter()
491                .map(|(_, request)| match request {
492                    CrossChainRequest::ConfirmUpdatedRecipient { latest_height, .. } => {
493                        latest_height.0
494                    }
495                    other => panic!("not a confirmation: {other:?}"),
496                })
497                .collect()
498        }
499    }

562    /// A queue holds at most one pending request: while one is in flight, further requests
563    /// for the same queue overwrite each other and only the last one is delivered. Senders
564    /// must therefore not rely on every request they submit reaching its target.
565    #[tokio::test(start_paused = true)]

567        let handler = Handler::new().gated();
568        let (mut sender, _task) = spawn_simple_forwarder(&handler);
569
570        send(&mut sender, confirm(1, 2, 1), 0).await;
571        settle().await;
572        assert_eq!(handler.tags(), vec![1], "the first request is in flight");
573
574        send(&mut sender, confirm(1, 2, 2), 0).await;
575        send(&mut sender, confirm(1, 2, 3), 0).await;
576        settle().await;
577        assert_eq!(handler.tags(), vec![1], "nothing else has started yet");
578
579        handler.release(1);
580        settle().await;
581        assert_eq!(handler.tags(), vec![1, 3], "request 2 was superseded by 3");
582    }

743    /// The wait before the n-th retry is `retry_delay * n`, capped at `max_backoff`.
744    #[tokio::test(start_paused = true)]
745    async fn test_retry_delay_grows_linearly_up_to_the_maximum() {
746        let handler = Handler::new().failing(usize::MAX);

754        );
755
756        let start = tokio::time::Instant::now();
