88impl ServerContext {
89    fn make_shard_state<S>(
90        &self,
91        local_ip_addr: &str,
92        shard_id: ShardId,
93        storage: S,
94        block_export: Option<BlockExportHandle>,
95    ) -> (WorkerState<S>, ShardId, ShardConfig)
96    where
97        S: Storage + Clone + Send + Sync + 'static,
98    {
99        let shard = self.server_config.internal_network.shard(shard_id);
100        info!("Shard booted on {}", shard.host);
101        info!(
102            "Public key: {}",
103            self.server_config.validator_secret.public()
104        );
105        let config = ChainWorkerConfig {
106            nickname: format!("Shard {} @ {}:{}", shard_id, local_ip_addr, shard.port),
107            key_pair: Some(Arc::new(self.server_config.validator_secret.copy())),
108            allow_inactive_chains: true,
109            block_time_grace_period: self.block_time_grace_period,
110            ttl: util::non_zero_duration(self.chain_worker_ttl),
111            chain_info_max_received_log_entries: self.chain_info_max_received_log_entries,
112            cross_chain_message_chunk_limit: self.cross_chain_message_chunk_limit,
113            cross_chain_batch_size_limit: self.cross_chain_batch_size_limit,
114            block_cache_size: self.block_cache_size,
115            execution_state_cache_size: self.execution_state_cache_size,
116            allow_revert_confirm: self.allow_revert_confirm,
117            reset_on_corrupted_chain_state: self
118                .reset_on_corrupted_chain_state_mins
119                .map(|m| Duration::from_secs(m * 60)),
120            recovery_whitelist: self.recovery_whitelist.clone(),
121            ..ChainWorkerConfig::default()
122        };
123        let mut state = WorkerState::new(storage, config, None);
124        if let Some(handle) = block_export {
125            state = state.with_block_export(handle);
126        }
127        (state, shard_id, shard.clone())
128    }
129
130    /// Spawns the process-wide block export queue, if export is enabled. One per process: every
131    /// shard in it hands blocks to the same queue, so a peer costs one connection pool and one
132    /// rate-limit window rather than one per shard.
133    fn spawn_block_export<S>(&self, storage: S) -> Option<BlockExportHandle>
134    where
135        S: Storage + Clone + Send + Sync + 'static,
136    {
137        let export_config = self.block_export_config.clone()?;
138        let own_public_key = self.server_config.validator_secret.public();
139        let node_options = self.block_export_node_options;
140        // Both arms get the destination's committee address and differ only in who dials it. In
141        // neither do the retry fields of `node_options` take effect — the relay pool drops them,
142        // the direct provider gets zeros — because retrying is the export queue's job.
143        let handle = match self.block_export_transport {
144            BlockExportTransport::Relay => {
145                let internal_network = &self.server_config.internal_network;
146                // All of this validator's proxies, rotated over by the provider: each export
147                // leaves through exactly one of them, but successive destinations draw different
148                // ones so the egress is spread rather than landing entirely on the first.
149                let relay_addresses = internal_network
150                    .proxies
151                    .iter()
152                    .map(|proxy| proxy.internal_address(&internal_network.protocol))
153                    .collect::<Vec<_>>();
154                assert!(
155                    !relay_addresses.is_empty(),
156                    "relaying exported blocks needs at least one proxy to relay through, but this \
157                     validator's internal network configures none; use \
158                     `--block-export-transport direct` to send without a proxy",
159                );
160                spawn_block_export_queue(
161                    storage,
162                    Arc::new(grpc::RelayNodeProvider::new(relay_addresses, node_options)),
163                    export_config,
164                    Some(own_public_key),
165                )
166            }
167            BlockExportTransport::Direct => spawn_block_export_queue(
168                storage,
169                Arc::new(linera_rpc::NodeProvider::new(node_options)),
170                export_config,
171                Some(own_public_key),
172            ),
173        };
174        Some(handle)
175    }
176
177    #[cfg_attr(not(with_metrics), allow(unused_variables))]
178    fn spawn_simple<S>(
179        &self,
180        listen_address: &str,
181        states: Vec<(WorkerState<S>, ShardId, ShardConfig)>,
182        protocol: simple::TransportProtocol,
183        shutdown_signal: &CancellationToken,
184        enable_memory_profiling: bool,
185    ) -> JoinSet<()>
186    where
187        S: Storage + Clone + Send + Sync + 'static,
188    {
189        let mut join_set = JoinSet::new();
190        let handles = FuturesUnordered::new();
191
192        let internal_network = self
193            .server_config
194            .internal_network
195            .clone_with_protocol(protocol);
196
197        for (state, shard_id, shard) in states {
198            let internal_network = internal_network.clone();
199            let cross_chain_config = self.cross_chain_config.clone();
200            let listen_address = listen_address.to_owned();
201
202            #[cfg(with_metrics)]
203            if let Some(port) = shard.metrics_port {
204                monitoring_server::start_metrics(
205                    (listen_address.clone(), port),
206                    shutdown_signal.clone(),
207                    monitoring_server::MemoryProfiling::from(enable_memory_profiling),
208                    linera_service::init_metrics,
209                );
210            }
211
212            let server_handle = simple::Server::new(
213                internal_network,
214                listen_address,
215                shard.port,
216                state,
217                shard_id,
218                cross_chain_config,
219            )
220            .spawn(shutdown_signal.clone(), &mut join_set);
221
222            handles.push(
223                server_handle
224                    .join()
225                    .inspect_err(move |error| {
226                        error!("Error running server for shard {shard_id}: {error:?}")
227                    })
228                    .map(|_| ()),
229            );
230        }
231
232        join_set.spawn_task(handles.collect::<()>());
233
234        join_set
235    }
236
237    #[cfg_attr(not(with_metrics), allow(unused_variables))]
238    fn spawn_grpc<S>(
239        &self,
240        listen_address: &str,
241        states: Vec<(WorkerState<S>, ShardId, ShardConfig)>,
242        shutdown_signal: &CancellationToken,
243        enable_memory_profiling: bool,
244    ) -> JoinSet<()>
245    where
246        S: Storage + Clone + Send + Sync + 'static,
247    {
248        let mut join_set = JoinSet::new();
249        let handles = FuturesUnordered::new();
250
251        for (state, shard_id, shard) in states {
252            #[cfg(with_metrics)]
253            if let Some(port) = shard.metrics_port {
254                monitoring_server::start_metrics(
255                    (listen_address.to_string(), port),
256                    shutdown_signal.clone(),
257                    monitoring_server::MemoryProfiling::from(enable_memory_profiling),
258                    linera_service::init_metrics,
259                );
260            }
261
262            let server_handle = grpc::GrpcServer::spawn(
263                listen_address.to_string(),
264                shard.port,
265                state,
266                shard_id,
267                self.server_config.internal_network.clone(),
268                &self.cross_chain_config,
269                &self.notification_config,
270                shutdown_signal.clone(),
271                &mut join_set,
272            );
273
274            handles.push(
275                server_handle
276                    .join()
277                    .inspect_err(move |error| {
278                        error!("Error running server for shard {shard_id}: {error:?}")
279                    })
280                    .map(|_| ()),
281            );
282        }
283
284        join_set.spawn_task(handles.collect::<()>());
285
286        join_set
287    }
288
289    fn get_listen_address() -> String {
290        // Allow local IP address to be different from the public one.
291        "0.0.0.0".to_string()
292    }
293}

295#[async_trait]
296impl Runnable for ServerContext {
297    type Output = anyhow::Result<()>;
298
299    async fn run<S>(self, storage: S) -> anyhow::Result<()>
300    where
301        S: Storage + Clone + Send + Sync + 'static,
302    {
303        let shutdown_notifier = CancellationToken::new();
304        let listen_address = Self::get_listen_address();
305
306        tokio::spawn(listen_for_shutdown_signals(shutdown_notifier.clone()));
307
308        // Activate memory profiling once before per-shard start_metrics calls.
309        #[cfg(with_metrics)]
310        let enable_memory_profiling = {
311            let memory_profiling =
312                monitoring_server::MemoryProfiling::try_activate(self.enable_memory_profiling)
313                    .await;
314            memory_profiling == monitoring_server::MemoryProfiling::Enabled
315        };
316        #[cfg(not(with_metrics))]
317        let enable_memory_profiling = false;
318
319        // One export queue per process, shared by every shard this process runs.
320        let block_export = self.spawn_block_export(storage.clone());
321
322        // Run the server
323        let states = match self.shard {
324            Some(shard) => {
325                info!("Running shard number {}", shard);
326                vec![self.make_shard_state(&listen_address, shard, storage, block_export)]
327            }
328            None => {
329                info!("Running all shards");
330                let num_shards = self.server_config.internal_network.shards.len();
331                (0..num_shards)
332                    .map(|shard| {
333                        self.make_shard_state(
334                            &listen_address,
335                            shard,
336                            storage.clone(),
337                            block_export.clone(),
338                        )
339                    })
340                    .collect()
341            }
342        };
343
344        let mut join_set = match self.server_config.internal_network.protocol {
345            NetworkProtocol::Simple(protocol) => self.spawn_simple(
346                &listen_address,
347                states,
348                protocol,
349                &shutdown_notifier,
350                enable_memory_profiling,
351            ),
352            NetworkProtocol::Grpc(tls_config) => match tls_config {
353                TlsConfig::ClearText => self.spawn_grpc(
354                    &listen_address,
355                    states,
356                    &shutdown_notifier,
357                    enable_memory_profiling,
358                ),
359                TlsConfig::Tls => bail!("TLS not supported between proxy and shards."),
360            },
361        };
362
363        join_set.await_all_tasks().await;
364
365        Ok(())
366    }
367}

421struct ValidatorOptions {
422    /// Path to the file containing the server configuration of this Linera validator (including its secret key)
423    server_config_path: PathBuf,
424
425    /// The host of the validator (IP address or hostname)
426    host: String,
427
428    /// The port of the validator
429    port: u16,
430
431    /// The server configurations for the linera-exporter.
432    #[serde(default)]
433    block_exporters: Vec<ExporterServiceConfig>,
434
435    /// The network protocol for the frontend.
436    external_protocol: NetworkProtocol,
437
438    /// The network protocol for workers.
439    internal_protocol: NetworkProtocol,
440
441    /// The public name and the port of each of the shards
442    shards: Vec<ShardConfig>,
443
444    /// The name and the port of the proxies
445    proxies: Vec<ProxyConfig>,
446}
447
448fn make_server_config<R: CryptoRng>(
449    path: &Path,
450    rng: &mut R,
451    options: ValidatorOptions,
452) -> anyhow::Result<persistent::File<ValidatorServerConfig>> {
453    let validator_keypair = ValidatorKeypair::generate_from(rng);
454    let account_secret = AccountSecretKey::Ed25519(Ed25519SecretKey::generate_from(rng));
455    let public_key = validator_keypair.public_key;
456    let network = ValidatorPublicNetworkConfig {
457        protocol: options.external_protocol,
458        host: options.host,
459        port: options.port,
460    };
461    let internal_network = ValidatorInternalNetworkConfig {
462        public_key,
463        protocol: options.internal_protocol,
464        shards: options.shards,
465        block_exporters: options.block_exporters,
466        proxies: options.proxies,
467    };
468    let validator = ValidatorConfig {
469        network,
470        public_key,
471        account_key: account_secret.public(),
472    };
473    Ok(persistent::File::new(
474        path,
475        ValidatorServerConfig {
476            validator,
477            validator_secret: validator_keypair.secret_key,
478            internal_network,
479        },
480    )?)
481}

726fn main() {
727    let options = <ServerOptions as clap::Parser>::parse();
728
729    let mut runtime = if options.tokio_threads == Some(1) {
730        tokio::runtime::Builder::new_current_thread()
731    } else {
732        let mut builder = tokio::runtime::Builder::new_multi_thread();
733
734        if let Some(threads) = options.tokio_threads {
735            builder.worker_threads(threads);
736        }
737
738        builder
739    };
740
741    if let Some(blocking_threads) = options.tokio_blocking_threads {
742        runtime.max_blocking_threads(blocking_threads);
743    }
744
745    runtime
746        .enable_all()
747        .build()
748        .expect("Failed to create Tokio runtime")
749        .block_on(run(options))
750}

752/// Returns the log file name to use based on the [`ServerCommand`] that will run.
753fn otlp_exporter_endpoint_for(command: &ServerCommand) -> Option<&str> {
754    match command {
755        ServerCommand::Run {
756            otlp_exporter_endpoint,
757            ..
758        } => otlp_exporter_endpoint.as_deref(),
759        ServerCommand::Generate { .. } | ServerCommand::EditShards { .. } => None,
760    }
761}

763fn log_file_name_for(command: &ServerCommand) -> Cow<'static, str> {
764    match command {
765        ServerCommand::Run {
766            shard,
767            server_config_path,
768            ..
769        } => {
770            let server_config: ValidatorServerConfig =
771                util::read_json(server_config_path).expect("Failed to read server config");
772            let public_key = &server_config.validator.public_key;
773
774            if let Some(shard) = shard {
775                format!("validator-{public_key}-shard-{shard}")
776            } else {
777                format!("validator-{public_key}")
778            }
779            .into()
780        }
781        ServerCommand::Generate { .. } | ServerCommand::EditShards { .. } => "server".into(),
782    }
783}

785async fn run(options: ServerOptions) {
786    linera_service::tracing::opentelemetry::init(
787        &log_file_name_for(&options.command),
788        otlp_exporter_endpoint_for(&options.command),
789    );
790
791    #[cfg(with_metrics)]
792    let enable_memory_profiling = options.enable_memory_profiling();
793    match options.command {
794        ServerCommand::Run {
795            server_config_path,
796            storage_config,
797            common_storage_options,
798            cross_chain_config,
799            notification_config,
800            shard,
801            block_time_grace_period,
802            wasm_runtime,
803            chain_worker_ttl,
804            chain_info_max_received_log_entries,
805            cross_chain_message_chunk_limit,
806            cross_chain_batch_size_limit,
807            allow_revert_confirm,
808            reset_on_corrupted_chain_state_mins,
809            recovery_whitelist,
810            export_blocks_to_committee,
811            block_export_transport,
812            block_export_batch_size,
813            block_export_max_catch_up_blocks,
814            block_export_queue_size,
815            block_export_queue_bytes,
816            block_export_max_in_flight,
817            block_export_max_in_flight_total,
818            block_export_converged_retention,
819            block_export_idle_interval,
820            block_export_retry_delay,
821            block_export_max_retry_delay,
822            block_export_send_timeout,
823            block_export_recv_timeout,
824            otlp_exporter_endpoint: _,
825        } => {
826            linera_version::VERSION_INFO.log();
827
828            let server_config: ValidatorServerConfig =
829                util::read_json(&server_config_path).expect("Failed to read server config");
830
831            let job = ServerContext {
832                server_config,
833                cross_chain_config,
834                notification_config,
835                shard,
836                block_time_grace_period,
837                chain_worker_ttl,
838                block_cache_size: options.block_cache_size,
839                execution_state_cache_size: options.execution_state_cache_size,
840                chain_info_max_received_log_entries,
841                cross_chain_message_chunk_limit,
842                cross_chain_batch_size_limit,
843                allow_revert_confirm,
844                reset_on_corrupted_chain_state_mins,
845                recovery_whitelist: recovery_whitelist.map(HashSet::from_iter),
846                block_export_config: export_blocks_to_committee.then(|| {
847                    let config = BlockExportConfig {
848                        certificate_upload_batch_size: block_export_batch_size,
849                        queue_size: block_export_queue_size,
850                        queue_bytes: block_export_queue_bytes,
851                        max_in_flight_per_destination: block_export_max_in_flight,
852                        max_in_flight_total: block_export_max_in_flight_total,
853                        max_catch_up_blocks: block_export_max_catch_up_blocks,
854                        idle_catch_up_interval: block_export_idle_interval,
855                        retry_delay: block_export_retry_delay,
856                        max_retry_delay: block_export_max_retry_delay,
857                        converged_chain_retention: block_export_converged_retention,
858                    };
859                    config.check().expect("invalid block export configuration");
860                    config
861                }),
862                block_export_transport,
863                block_export_node_options: NodeOptions {
864                    send_timeout: block_export_send_timeout,
865                    recv_timeout: block_export_recv_timeout,
866                    ..NodeOptions::default()
867                },
868                #[cfg(with_metrics)]
869                enable_memory_profiling,
870            };
871            let wasm_runtime = wasm_runtime.with_wasm_default();
872            let store_config = storage_config
873                .add_common_storage_options(&common_storage_options)
874                .unwrap();
875            // Validators should not output contract logs.
876            let allow_application_logs = false;
877            let cache_sizes = common_storage_options.storage_cache_config();
878            store_config
879                .run_with_storage(wasm_runtime, allow_application_logs, cache_sizes, job)
880                .boxed()
881                .await
882                .unwrap()
883                .unwrap();
884        }
885
886        ServerCommand::Generate {
887            validators,
888            committee,
889            testing_prng_seed,
890        } => {
891            let mut config_validators = Vec::new();
892            let mut rng = Box::<dyn CryptoRng>::from(testing_prng_seed);
893            for options_path in validators {
894                let options_string = fs_err::tokio::read_to_string(options_path)
895                    .await
896                    .expect("Unable to read validator options file");
897                let options: ValidatorOptions = toml::from_str(&options_string)
898                    .unwrap_or_else(|_| panic!("Invalid options file format: \n {options_string}"));
899                let path = options.server_config_path.clone();
900                let mut server = make_server_config(&path, &mut rng, options)
901                    .expect("Unable to open server config file");
902                Persist::persist(&mut server)
903                    .await
904                    .expect("Unable to write server config file");
905                info!("Wrote server config {}", path.to_str().unwrap());
906                println!(
907                    "{},{}",
908                    server.validator.public_key, server.validator.account_key
909                );
910                config_validators.push(Persist::into_value(server).validator);
911            }
912            if let Some(committee) = committee {
913                let mut config = persistent::File::new(
914                    &committee,
915                    CommitteeConfig {
916                        validators: config_validators,
917                    },
918                )
919                .expect("Unable to open committee configuration");
920                Persist::persist(&mut config)
921                    .await
922                    .expect("Unable to write committee description");
923                info!("Wrote committee config {}", committee.to_str().unwrap());
924            }
925        }
926
927        ServerCommand::EditShards {
928            server_config_path,
929            num_shards,
930            host,
931            port,
932            metrics_port,
933        } => {
934            let mut server_config =
935                persistent::File::<ValidatorServerConfig>::read(&server_config_path)
936                    .expect("Failed to read server config");
937            let shards = generate_shard_configs(&num_shards, &host, &port, &metrics_port)
938                .expect("Failed to generate shard configs");
939            server_config.internal_network.shards = shards;
940            Persist::persist(&mut server_config)
941                .await
942                .expect("Failed to write updated server config");
943        }
944    }
945}

947fn generate_shard_configs(
948    num_shards: &str,
949    host: &str,
950    port: &str,
951    metrics_port: &Option<String>,
952) -> anyhow::Result<Vec<ShardConfig>> {
953    let mut shards = Vec::new();
954    let len = num_shards.len();
955    let num_shards = num_shards
956        .parse::<NonZeroU16>()
957        .context("Failed to parse the number of shards")?;
958    let pattern = "%".repeat(len);
959
960    for i in 1u16..=num_shards.into() {
961        let index = format!("{i:0len$}");
962        let host = host.replacen(&pattern, &index, 1);
963        let port = port
964            .replacen(&pattern, &index, 1)
965            .parse()
966            .context("Failed to decode port into an integers")?;
967        let metrics_port = metrics_port
968            .as_ref()
969            .map(|port| {
970                port.replacen(&pattern, &index, 1)
971                    .parse()
972                    .context("Failed to decode metrics port into an integers")
973            })
974            .transpose()?;
975        let shard = ShardConfig {
976            host,
977            port,
978            metrics_port,
979        };
980        shards.push(shard);
981    }
982    Ok(shards)
983}
